import json
from pathlib import Path
from typing import Any


class SeatDistributionRouter:
    """
    Routes seat-distribution table requests to dynamically
    detected prospectus page ranges.

    Page numbers are never hardcoded here.
    """

    SEAT_TERMS = (
        "seat distribution",
        "distribution of seats",
        "seat allocation",
        "seats available",
        "available seats",
        "number of seats",
        "how many seats",
        "total seats",
        "regular seats",
        "self finance seats",
        "self-finance seats",
    )

    SELF_FINANCE_TERMS = (
        "self finance",
        "self-finance",
        "self financing",
        "self-financing",
    )

    REGULAR_TERMS = (
        "regular",
        "merit seats",
        "open merit seats",
    )

    def route(
        self,
        question: str,
        sections_path: Path | None,
    ) -> dict[str, Any] | None:

        normalized = " ".join(
            question.casefold().split()
        )

        if not self._is_seat_request(normalized):
            return None

        sections = self._load_sections(
            sections_path
        )

        if not sections:
            return None

        target_types = self._target_section_types(
            normalized
        )

        matched_sections = [
            section
            for section in sections
            if section.get("section_type")
            in target_types
        ]

        if not matched_sections:
            return None

        document = self._document_id(
            sections_path
        )

        page_references = []

        for section in matched_sections:
            section_type = section[
                "section_type"
            ]

            label = self._section_label(
                section_type
            )

            for page_number in range(
                section["start_page"],
                section["end_page"] + 1,
            ):
                page_references.append(
                    {
                        "document": document,
                        "page_number": page_number,
                        "page_path": None,
                        "page_url": (
                            "/api/v1/pages/"
                            f"{document}/"
                            f"{page_number}"
                        ),
                        "reason": (
                            f"View {label} seat "
                            "distribution."
                        ),
                    }
                )

        answer = self._build_answer(
            matched_sections
        )

        return {
            "question": question,
            "answer": answer,
            "status": "seat_distribution_pages",
            "sources": [],
            "page_references": page_references,
            "retrieval": {
                "route": "seat_distribution",
                "vector_chunks": [],
                "vector_chunk_count": 0,
            },
        }

    def _is_seat_request(
        self,
        normalized: str,
    ) -> bool:

        return any(
            term in normalized
            for term in self.SEAT_TERMS
        )

    def _target_section_types(
        self,
        normalized: str,
    ) -> set[str]:

        wants_self_finance = any(
            term in normalized
            for term in self.SELF_FINANCE_TERMS
        )

        wants_regular = any(
            term in normalized
            for term in self.REGULAR_TERMS
        )

        if wants_self_finance and not wants_regular:
            return {
                "seat_distribution_self_finance"
            }

        if wants_regular and not wants_self_finance:
            return {
                "seat_distribution_regular"
            }

        return {
            "seat_distribution_regular",
            "seat_distribution_self_finance",
        }

    @staticmethod
    def _load_sections(
        sections_path: Path | None,
    ) -> list[dict[str, Any]]:

        if sections_path is None:
            return []

        if not sections_path.exists():
            return []

        try:
            data = json.loads(
                sections_path.read_text(
                    encoding="utf-8"
                )
            )
        except (
            OSError,
            json.JSONDecodeError,
        ):
            return []

        sections = data.get("sections", [])

        if not isinstance(sections, list):
            return []

        return sections

    @staticmethod
    def _document_id(
        sections_path: Path,
    ) -> str:

        try:
            data = json.loads(
                sections_path.read_text(
                    encoding="utf-8"
                )
            )

            document_id = data.get(
                "document_id"
            )

            if document_id:
                return str(document_id)

        except (
            OSError,
            json.JSONDecodeError,
        ):
            pass

        return sections_path.stem

    @staticmethod
    def _section_label(
        section_type: str,
    ) -> str:

        if (
            section_type
            == "seat_distribution_regular"
        ):
            return "regular"

        return "self-finance"

    def _build_answer(
        self,
        sections: list[dict[str, Any]],
    ) -> str:

        labels = [
            self._section_label(
                section["section_type"]
            )
            for section in sections
        ]

        if len(labels) == 1:
            return (
                f"The {labels[0]} seat distribution "
                "is available in the prospectus. "
                "Use the page links below to view "
                "the official seat tables."
            )

        return (
            "The regular and self-finance seat "
            "distributions are available in the "
            "prospectus. Use the page links below "
            "to view the official seat tables."
        )

export const API_BASE_URL: string =
  import.meta.env.VITE_API_BASE_URL ||
  "http://127.0.0.1:8000";

export class ApiError extends Error {
  status?: number;

  constructor(
    message: string,
    status?: number,
  ) {
    super(message);
    this.name = "ApiError";
    this.status = status;
  }
}

interface RequestOptions {
  method?:
    | "GET"
    | "POST"
    | "PUT"
    | "PATCH"
    | "DELETE";
  body?: unknown;
  headers?: Record<string, string>;
}

export async function apiRequest<T>(
  path: string,
  options: RequestOptions = {},
): Promise<T> {
  const {
    method = "GET",
    body,
    headers = {},
  } = options;

  let response: Response;

  try {
    response = await fetch(
      `${API_BASE_URL}${path}`,
      {
        method,
        headers: {
          "Content-Type": "application/json",
          ...headers,
        },
        body:
          body !== undefined
            ? JSON.stringify(body)
            : undefined,
      },
    );
  } catch {
    throw new ApiError(
      "Could not reach the ProspectusAI server. " +
        "Please check your connection and try again.",
    );
  }

  if (!response.ok) {
    let message =
      `Request failed with status ${response.status}`;

    try {
      const errorBody = await response.json();

      if (
        typeof errorBody?.detail === "string"
      ) {
        message = errorBody.detail;
      } else if (
        typeof errorBody?.message === "string"
      ) {
        message = errorBody.message;
      }
    } catch {
      // Keep the default message.
    }

    throw new ApiError(
      message,
      response.status,
    );
  }

  const text = await response.text();

  if (!text) {
    return undefined as T;
  }

  return JSON.parse(text) as T;
}

export function buildPageUrl(
  pageUrl: string,
): string {
  if (/^https?:\/\//i.test(pageUrl)) {
    return pageUrl;
  }

  return (
    `${API_BASE_URL}` +
    `${pageUrl.startsWith("/") ? "" : "/"}` +
    pageUrl
  );
}
