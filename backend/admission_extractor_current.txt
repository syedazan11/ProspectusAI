import re

from src.processing.prospectus.admission_record import (
    AdmissionRecord,
)
from dataclasses import dataclass


@dataclass(slots=True)
class VisualLine:

    text: str

    items: list[dict]

    top: float

    left: float

    right: float

    height: float


@dataclass(slots=True)
class LogicalLine:

    text: str

    lines: list[VisualLine]

    left: float

    top: float

class AdmissionExtractor:
    """
    Extract structured admission information.
    """

    SUBSECTION_PATTERN = re.compile(
        r"""
        ^
        (?P<number>
            \d+
            (?:\.\d+)*
            \.?
            (?:\s*\([a-z]\))?
        )
        \s+
        (?P<title>.+?)
        \s*$
        """,
        re.IGNORECASE | re.VERBOSE,
    )


    def _build_visual_lines(
        self,
        layout_items: list[dict],
    ) -> list[VisualLine]:

        buckets: dict[int, list[dict]] = {}

        for item in layout_items:

            text = item.get(
                "text",
                "",
            ).strip()
    
            if not text:
                continue
    
            key = round(
                item.get(
                    "top",
                    0,
                )
                / 6
            )

            buckets.setdefault(
                key,
                [],
            ).append(item)

        visual_lines: list[VisualLine] = []

        for key in sorted(buckets):

            row = sorted(
                buckets[key],
                key=lambda x: x.get(
                    "x",
                    0,
                ),
            )

            text = " ".join(
                cell["text"].strip()
                for cell in row
            ).strip()

            if not text:
                continue

            left = min(
                cell.get(
                    "x",
                    0,
                )
                for cell in row
            )

            right = max(
                cell.get(
                    "x",
                    0,
                )
                for cell in row
            )

            top = sum(
                cell.get(
                    "top",
                    0,
                )
                for cell in row
            ) / len(row)

            height = max(
                cell.get(
                    "height",
                    0,
                )
                for cell in row
            )

            visual_lines.append(
                VisualLine(
                    text=text,
                    items=row,
                    left=left,
                    right=right,
                    top=top,
                    height=height,
                )
            )

        return visual_lines

    def _merge_wrapped_lines(
        self,
        lines: list[VisualLine],
    ) -> list[LogicalLine]:

        logical: list[LogicalLine] = []

        for line in lines:

            if not logical:

                logical.append(
                    LogicalLine(
                        text=line.text,
                        lines=[line],
                        left=line.left,
                        top=line.top,
                    )
                )

                continue

            previous = logical[-1]

            previous_line = previous.lines[-1]

            vertical_gap = (
                line.top
                - previous_line.top
            )

            indent_gap = abs(
                line.left
                - previous.left
            )

            starts_new_block = bool(
                re.match(
                    r"""
                    ^
                    (
                        \d+(?:\.\d+)+      # 4.2.1
                        |
                        \(?[a-z]\)         # (a)
                        |
                        [ivxlcdm]+\)       # ii)
                        |
                        •
                        |
                        -
                    )
                    """,
                    line.text,
                    re.IGNORECASE | re.VERBOSE,
                )
            )

            continuation = (
                not starts_new_block
                and vertical_gap < 18
                and indent_gap < 35
            )

            if continuation:

                previous.text += (
                    " "
                    + line.text
                )

                previous.lines.append(
                    line
                )

            else:

                logical.append(
                    LogicalLine(
                        text=line.text,
                        lines=[line],
                        left=line.left,
                        top=line.top,
                    )
                )

        return logical
    
    def _split_columns(
        self,
        lines: list[VisualLine],
    ) -> list[list[VisualLine]]:
        """
        Split a page into reading columns.

        Returns one or two columns ordered
        from left to right.
        """

        if not lines:
            return []

        xs = sorted(
            (line.left + line.right) / 2
            for line in lines
        )

        if len(xs) < 6:
            return [sorted(
                lines,
                key=lambda x: x.top,
            )]

        gaps = []

        for i in range(len(xs) - 1):

            gap = xs[i + 1] - xs[i]

            gaps.append(
                (
                    gap,
                    i,
                )
            )

        largest_gap, index = max(gaps)

        # No meaningful split.
        if largest_gap < 120:

            return [
                sorted(
                    lines,
                    key=lambda x: x.top,
                )
            ]

        boundary = (
            xs[index]
            + xs[index + 1]
        ) / 2

        left = []
        right = []

        for line in lines:

            center = (
                line.left + line.right
            ) / 2

            if center < boundary:
                left.append(line)
            else:
                right.append(line)

        columns = []

        if left:
            columns.append(
                sorted(
                    left,
                    key=lambda x: x.top,
                )
            )

        if right:
            columns.append(
                sorted(
                    right,
                    key=lambda x: x.top,
                )
            )

        return columns
    
    def _match_subsection(
        self,
        text: str,
    ) -> re.Match | None:

        return self.SUBSECTION_PATTERN.match(
            text.strip()
        )

    def _clean_text(
        self,
        text: str,
    ) -> str:

        text = re.sub(
            r"\s+",
            " ",
            text,
        )

        text = text.replace(
            "•",
            ""
        )

        text = text.strip()

        return text
    
    def _extract_logical_lines(
        self,
        *,
        logical_lines: list[LogicalLine],
        page_number: int,
        heading: str,
    ) -> list[AdmissionRecord]:

        print("ENTERED _extract_logical_lines")

        records: list[AdmissionRecord] = []

        current_title: str | None = None
        current_number: str | None = None

        current_lines: list[str] = []
        intro_lines: list[str] = []

        def flush():

            nonlocal current_title
            nonlocal current_number
            nonlocal current_lines

            if not current_title:
                return

            body = "\n".join(
                current_lines
            ).strip()

            source_text = current_title

            if body:
                source_text += (
                    "\n"
                    + body
                )

            records.append(
                AdmissionRecord(
                    title=current_title,
                    category=heading,
                    details={},
                    source_page=page_number,
                    source_heading=heading,
                    source_text=source_text,
                )
            )

            current_title = None
            current_number = None
            current_lines = []

        for logical in logical_lines:

            text = self._clean_text(
                logical.text
            )
            print(repr(text))

            if len(text) < 3:
                continue

            if text.lower() in {
                "chapter",
                "contents",
                "prospectus",
            }:
                continue

            if not text:
                continue

            match = self._match_subsection(text)

            if match:
                flush()
                current_number = match.group(
                    "number"
                )
                current_title = self._clean_text(
                    match.group("title")
                )

                current_lines = []

                if intro_lines:

                    current_lines.extend(
                        intro_lines
                    )

                    intro_lines.clear()

                continue

            if current_title is None:

                intro_lines.append(
                    text
                )
                continue

            current_lines.append(text)

        flush()

        return records

    def extract(
        self,
        pages: list[dict],
    ) -> list[AdmissionRecord]:

        print(f"extract() received {len(pages)} pages")

        records: list[AdmissionRecord] = []

        for page in pages:

            print(
                f"Page {page['page_number']} "
                f"sections={len(page.get('sections', []))}"
            )

            page_number = page["page_number"]

            for section in page.get(
                "sections",
                [],
            ):

                heading = (
                    section.get(
                        "heading",
                        "",
                    ).strip()
                )

                layout_items = (
                    section.get(
                        "layout_items"
                    )
                    or []
                )

                # Fallback for older documents.
                if not layout_items:

                    records.extend(
                        self._extract_section(
                            section=section,
                            page_number=page_number,
                            heading=heading,
                        )
                    )

                    continue

                print(
                    f"Page {page_number} | Heading: {heading} | layout_items={len(layout_items)}"
                )

                visual_lines = (
                    self._build_visual_lines(
                        layout_items
                    )
                )

                columns = (
                    self._split_columns(
                        visual_lines
                    )
                )

                print(f"Columns: {len(columns)}")

                for column in columns:

                    print(f"Column lines: {len(column)}")

                    logical_lines = (
                        self._merge_wrapped_lines(
                            column
                        )
                    )

                    records.extend(
                        self._extract_logical_lines(
                            logical_lines=logical_lines,
                            page_number=page_number,
                            heading=heading,
                        )
                    )

        return records
